 def project(self,v,budget):
  # Field projection, never drop a whole source result because metadata/notices are large.
  if len(json.dumps(v,ensure_ascii=False))<=budget:return v
  if budget<150:return {'omission':'remaining channel transport budget; raw retained'}
  if isinstance(v,dict):
   out={};order=sorted(v,key=lambda k:(k not in ('content','source','files','records'),k));remaining=budget-100
   for k in order:
    if remaining<80:out['projection_omits_remaining_fields']=True;break
    out[k]=self.project(v[k],remaining);remaining-=len(json.dumps({k:out[k]},ensure_ascii=False))
   out['projection_partial']=True;return out
  if isinstance(v,list):
   out=[];remaining=budget-80
   for item in v:
    if remaining<100:out.append({'omission':'remaining records; raw retained'});break
    projected=self.project(item,remaining);out.append(projected);remaining-=len(json.dumps(projected,ensure_ascii=False))
   return out
  if isinstance(v,str):return {'omission':'field exceeds remaining transport; raw retained','characters':len(v)}
  return v
